# 최소비용 최대유량 (MCMF)
Min-cost Max-Flow

간선에 용량 뿐만 아닌 비용(Cost)까지 추가됨   
기존 방식 (디닉 등등)은 적용이 불가능  

유량을 최대한 흘려주되, 그 비용을 최소한으로 만드는 것  

[블로그](https://m.blog.naver.com/kks227/220810623254)  
[간단한 내용](https://www.crocus.co.kr/1090)

## 원리

에드몬드 카프 알고리즘과 비슷하게  
"매번" 경로를 찾아 유량을 흘리고 더이상 경로가 없을 때까지 반복하는 것  

이때 매번 찾는 **최단경로** 는 소스에서 싱크로 가는 최소 비용의 거리.  
매번 최단 경로를 찾고 이 경로로 흐를 수 있는 최대유량을 흘러보낸 뒤 결과에는(비용 합)에는 해당 경로 비용 합 X 최대유량 만큼 더함  


다만 여기선 "음의 가중치" 그래프에도 동작해야함  
비용 역시 **역방향 간선** 이라는 것도 적용해야 하기 때문  

따라서 주의사항도 "비용" 도 역방향 간선 적용을 해야 한다는 것!  

## SPFA
[관련 설명](https://huisam.tistory.com/entry/Bellman-Ford-SPFA)  
Shortest path faster Algorithm  
기본 아이디어는 벨만-포드 알고리즘  
보기에는 마치 "다익스트라" 알고리즘에서 우선순위 큐에서 그냥 큐를 쓰는 것처럼 보임  


## 시간복잡도

SPFA 의 시간복잡도의 최대 상한선은 $O(VE)$ 임 (벨만 - 포드가 베이스이기때문)  
따라서 최약 시간복잡도는 $O(VEf)$ 임  
하지만 실질적으로 SPFA의 시간복잡도의 기댓값은 $O(V + E)$임  (벨만 - 포드의 기댓값은 $VE$ 인 것 보면... )




## 코드예시

```c++
#include <iostream>
#include <vector>
#include <algorithm>
#include <queue>
#include <cstring>
using namespace std;
typedef pair<int, int> pii;
int n, m;
vector<int> adj[900]; // to, cost
int src = 0, sink;
// s : 0, e  : n + m + 1;
const int inf = 1e9;

int c[900][900], f[900][900], cost[900][900]; // 용량, 유량, 비용
int dist[900], prv[900]; // spfa 전용
bool v[900]; //queue에 포함되었는지 판단여부

int main(void) {
    ios_base::sync_with_stdio(false); cin.tie(0); cout.tie(0);
    int to, wcost;
    int total_flow = 0;
    int total_cost = 0;
    // input
    cin >> n >> m;
    sink = n + m + 1;

    // 연결

    for (int person = 1; person <= n; ++person) {
        // src -> person
        adj[src].push_back(person);
        adj[person].push_back(src);
        c[src][person] = 1;

        //person -> work(to)
        int wcnt; cin >> wcnt;
        while (wcnt--)
        {
            cin >> to >> wcost;
            to += n;
            adj[person].push_back(to);
            adj[to].push_back(person);

            // 비용 역방향에 대한 처리
            cost[person][to] = wcost;
            cost[to][person] = -wcost;

            // 용량
            c[person][to] = 1;
        }
    }

    // work -> sink
    for (int w = n+1; w <= n+m; ++w) {
        adj[w].push_back(sink);
        adj[sink].push_back(w);
        c[w][sink] = 1;
    }

    // mcmf

    while (1)
    {
        // 초기화 + src처리
        memset(prv, -1, sizeof(prv));
        fill(dist, dist+sink+1, inf);
        memset(v, 0, sizeof(v));

        prv[src] = src;
        dist[src] = 0;
        v[src] = true;
        queue<int> q;
        q.push(src);

        // spfa

        while (!q.empty())
        {
            int loc = q.front(); q.pop(); v[loc] = false;
            for (int nxt : adj[loc]) {
                if (c[loc][nxt] - f[loc][nxt] > 0 && dist[nxt] > dist[loc] + cost[loc][nxt]) {
                    dist[nxt] = dist[loc] + cost[loc][nxt];
                    prv[nxt] = loc;

                    if (!v[nxt]) {
                        v[nxt] = true;
                        q.push(nxt);
                    }
                }
            }
        }

        if (prv[sink] == -1) break;


        int max_allowflow = inf;
        for (int i = sink; i != src; i=prv[i]) {
            max_allowflow = min(max_allowflow, c[prv[i]][i] - f[prv[i]][i]);
        }

        for (int i = sink; i != src; i = prv[i]) {
            total_cost += cost[prv[i]][i] * max_allowflow;
            f[prv[i]][i] += max_allowflow;
            f[i][prv[i]] -= max_allowflow;
        }
        total_flow++;
    }

    cout << total_flow << '\n' << total_cost;

    return 0;
}

```
